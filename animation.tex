\documentclass[tikz, border=2mm]{standalone}
\usepackage{pgfplots}
\usepackage{multido}
\pgfplotsset{compat=1.18}

% Creates an explicit structural frame environment for multi-page standalone outputs
\standaloneenv{tikzpicture}

\begin{document}

\multido{\timeStep=1+1}{30}{
  \begin{tikzpicture}
    \begin{axis}
        [
        title={Live Data Feed (Time-Series Animation)},
        xlabel={Time $t$ (ms)},
        ylabel={Signal Output},
        xmin=0, xmax=35,
        ymin=-1.2, ymax=1.2,
        grid=major,
        width=10cm, height=6cm
        ]
        
        % Background reference curve
        \addplot[gray!30, dashed, domain=0:35, samples=100] {sin(deg(x/2))};

        % Growing time-series trajectory
        \addplot[blue, thick, domain=0:\timeStep, samples=50] {sin(deg(x/2))};
        
        % Frontier indicator point
        \addplot[red, mark=*, mark size=2pt] coordinates {(\timeStep, {sin(deg(\timeStep/2))})};
        
    \end{axis}
  \end{tikzpicture}
}

\end{document}
